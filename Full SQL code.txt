/* We had three results from searching for the plate numbers as seen in the drivers
license table and as found 3 likelihoods but they do not have names nor things that could help
us trace them hence we can join person table to license table so we can pin
point who has what id */

CREATE TABLE full_data AS
(SELECT dl.age, dl.height, dl.hair_color, dl.gender, dl.plate_number, 
dl.car_make, dl.car_model, p.name, p.ssn, p.address_street_name, 
p.id
FROM drivers_license AS dl
LEFT JOIN person AS p
ON dl.id = p.license_id
)

SELECT * FROM full_data
WHERE plate_number LIKE 'H42W%'
OR
plate_number LIKE '%H42W%'
OR
plate_number LIKE '%H42W'
